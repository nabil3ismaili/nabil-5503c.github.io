        let allOrders = [];

        window.showTab = function(tabName) {
            document.querySelectorAll('.tab-content').forEach(t => t.classList.remove('active'));
            document.querySelectorAll('.nav-item').forEach(n => n.classList.remove('active'));
            document.getElementById('tab-' + tabName).classList.add('active');
            event.currentTarget.classList.add('active');
        };

        async function loadStats() {
            try {
                const ordersSnap = await getDocs(collection(db, 'orders'));
                const productsSnap = await getDocs(collection(db, 'products'));
                
                allOrders = [];
                let newCount = 0;
                let revenue = 0;
                
                ordersSnap.forEach(d => {
                    const o = d.data();
                    allOrders.push({ id: d.id, ...o });
                    if (o.status === 'new') newCount++;
                    revenue += o.total || 0;
                });

                document.getElementById('statTotalOrders').textContent = allOrders.length;
                document.getElementById('statNewOrders').textContent = newCount;
                document.getElementById('statRevenue').textContent = revenue.toLocaleString();
                document.getElementById('statProducts').textContent = productsSnap.size;

                const badge = document.getElementById('orderBadge');
                if (newCount > 0) {
                    badge.textContent = newCount;
                    badge.style.display = 'inline-block';
                    badge.classList.add('danger');
                }

                renderRecentOrders();
                renderAllOrders();
            } catch (err) {
                showToast('Error loading data. Check Firebase config.', 'error');
            }
        }

        function renderRecentOrders() {
            const tbody = document.getElementById('recentOrdersTable');
            if (allOrders.length === 0) {
                tbody.innerHTML = '<tr><td colspan="6" style="text-align:center; color: var(--light-gray); padding: 2rem;">No orders yet</td></tr>';
                return;
            }
            const recent = allOrders.slice(0, 5);
            tbody.innerHTML = recent.map(o => `
                <tr>
                    <td><code style="font-size: 0.75rem; color: var(--gold);">${o.id.substring(0, 8)}...</code></td>
                    <td>${o.customer?.name || 'N/A'}</td>
                    <td>${o.items?.length || 0} items</td>
                    <td style="color: var(--gold); font-weight: 500;">${o.total?.toLocaleString() || 0} DZD</td>
                    <td><span class="status-badge status-${o.status || 'new'}">${o.status || 'new'}</span></td>
                    <td style="color: var(--light-gray); font-size: 0.8rem;">${formatDate(o.createdAt)}</td>
                </tr>
            `).join('');
        }

        function renderAllOrders() {
            const tbody = document.getElementById('allOrdersTable');
            if (allOrders.length === 0) {
                tbody.innerHTML = '<tr><td colspan="9" style="text-align:center; color: var(--light-gray); padding: 2rem;">No orders yet</td></tr>';
                return;
            }
            tbody.innerHTML = allOrders.map(o => `
                <tr>
                    <td><code style="font-size: 0.75rem; color: var(--gold);">${o.id.substring(0, 12)}</code></td>
                    <td>${o.customer?.name || 'N/A'}</td>
                    <td>${o.customer?.phone || 'N/A'}</td>
                    <td>${o.items?.length || 0} items</td>
                    <td style="color: var(--gold); font-weight: 500;">${o.total?.toLocaleString() || 0} DZD</td>
                    <td>${o.customer?.wilaya || 'N/A'}</td>
                    <td>
                        <select class="status-select" onchange="updateOrderStatus('${o.id}', this.value)">
                            <option value="new" ${o.status === 'new' ? 'selected' : ''}>New</option>
                            <option value="processing" ${o.status === 'processing' ? 'selected' : ''}>Processing</option>
                            <option value="completed" ${o.status === 'completed' ? 'selected' : ''}>Completed</option>
                        </select>
                    </td>
                    <td style="color: var(--light-gray); font-size: 0.8rem;">${formatDate(o.createdAt)}</td>
                    <td>
                        <button class="btn btn-sm btn-outline" onclick="viewOrder('${o.id}')">View</button>
                    </td>
                </tr>
            `).join('');
        }

        window.updateOrderStatus = async function(orderId, status) {
            try {
                await updateDoc(doc(db, 'orders', orderId), { status });
                showToast('Status updated!');
                loadStats();
            } catch (err) {
                showToast('Error updating status', 'error');
            }
        };

        window.viewOrder = function(orderId) {
            const order = allOrders.find(o => o.id === orderId);
            if (!order) return;
            
            const body = document.getElementById('orderModalBody');
            body.innerHTML = `
                <div class="detail-row"><span class="detail-label">Order ID</span><span class="detail-value">${order.id}</span></div>
                <div class="detail-row"><span class="detail-label">Customer</span><span class="detail-value">${order.customer?.name}</span></div>
                <div class="detail-row"><span class="detail-label">Phone</span><span class="detail-value">${order.customer?.phone}</span></div>
                <div class="detail-row"><span class="detail-label">Email</span><span class="detail-value">${order.customer?.email || 'N/A'}</span></div>
                <div class="detail-row"><span class="detail-label">Wilaya</span><span class="detail-value">${order.customer?.wilaya}</span></div>
                <div class="detail-row"><span class="detail-label">Address</span><span class="detail-value">${order.customer?.address}</span></div>
                ${order.customer?.notes ? `<div class="detail-row"><span class="detail-label">Notes</span><span class="detail-value">${order.customer.notes}</span></div>` : ''}
                <div class="detail-row"><span class="detail-label">Date</span><span class="detail-value">${formatDate(order.createdAt)}</span></div>
                <div class="detail-row"><span class="detail-label">Status</span><span class="detail-value"><span class="status-badge status-${order.status}">${order.status}</span></span></div>
                <div class="items-list">
                    <h4>Items Ordered</h4>
                    <ul>
                        ${order.items?.map(i => `<li>${i.name} x ${i.qty} — ${(i.price * i.qty).toLocaleString()} DZD</li>`).join('') || ''}

                <div class="detail-row" style="border-top: 2px solid var(--gold); margin-top: 1rem; padding-top: 1rem;">
                    <span class="detail-label" style="font-size: 1rem; color: var(--white);">Total</span>
                    <span class="detail-value" style="font-family: 'Playfair Display', serif; font-size: 1.5rem; color: var(--gold);">${order.total?.toLocaleString() || 0} DZD</span>
                </div>
            `;
            document.getElementById('orderModal').classList.add('open');
        };

        window.closeOrderModal = function() {
            document.getElementById('orderModal').classList.remove('open');
        };

        async function loadProducts() {
            try {
                const snap = await getDocs(collection(db, 'products'));
                allProducts = [];
                const tbody = document.getElementById('productsTable');
                
                if (snap.empty) {
                    tbody.innerHTML = '<tr><td colspan="6" style="text-align:center; color: var(--light-gray); padding: 2rem;">No products yet. Add your first one!</td></tr>';
                    return;
                }

                snap.forEach(d => {
                    const p = d.data();
                    allProducts.push({ id: d.id, ...p });
                });

                tbody.innerHTML = allProducts.map(p => `
                    <tr>
                        <td>
                            <div class="product-thumb">
                                ${p.image ? `<img src="${p.image}" alt="">` : '🛍️'}
                            </div>
                        </td>
                        <td>
                            <strong>${p.name}</strong>
                            <p style="font-size: 0.8rem; color: var(--light-gray); margin-top: 0.3rem;">${p.description?.substring(0, 50) || ''}...</p>
                        </td>
                        <td>${p.category || 'General'}</td>
                        <td style="color: var(--gold); font-weight: 500;">${parseInt(p.price).toLocaleString()} DZD</td>
                        <td><span class="status-badge status-new">Active</span></td>
                        <td>
                            <button class="btn btn-sm btn-danger" onclick="deleteProduct('${p.id}')">Delete</button>
                        </td>
                    </tr>
                `).join('');
            } catch (err) {
                showToast('Error loading products', 'error');
            }
        }

        window.deleteProduct = async function(productId) {
            if (!confirm('Delete this product?')) return;
            try {
                await deleteDoc(doc(db, 'products', productId));
                showToast('Product deleted!');
                loadProducts();
                loadStats();
            } catch (err) {
                showToast('Error deleting product', 'error');
            }
        };

        document.getElementById('addProductForm').addEventListener('submit', async (e) => {
            e.preventDefault();
            const btn = e.target.querySelector('button[type="submit"]');
            btn.textContent = 'Adding...';
            
            try {
                await addDoc(collection(db, 'products'), {
                    name: document.getElementById('newProdName').value,
                    price: parseInt(document.getElementById('newProdPrice').value),
                    category: document.getElementById('newProdCategory').value,
                    badge: document.getElementById('newProdBadge').value,
                    description: document.getElementById('newProdDesc').value,
                    image: document.getElementById('newProdImage').value,
                    createdAt: new Date().toISOString()
                });
                showToast('Product added successfully!');
                e.target.reset();
                loadProducts();
                loadStats();
            } catch (err) {
                showToast('Error adding product', 'error');
            } finally {
                btn.textContent = 'Add Product';
            }
        });

        function showToast(msg, type = 'success') {
            const toast = document.getElementById('toast');
            toast.textContent = msg;
            toast.className = 'toast ' + type + ' show';
            setTimeout(() => toast.classList.remove('show'), 3000);
        }

        function formatDate(iso) {
            if (!iso) return 'N/A';
            const d = new Date(iso);
            return d.toLocaleDateString('en-GB', { day: '2-digit', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit' });
        }

        window.refreshData = function() {
            loadStats();
            loadProducts();
            showToast('Data refreshed!');
        };

        loadStats();
        loadProducts();
    </script>
